# 3. Data Application
We apply three modeling approahces (two-stage method, Bayesian hierarchical model, and spatial Bayesian hierarchical model) to empirical dataset. The dataset includes daily death counts, ozone concentration (8 hour moving average), and temperature (Kelvin) for 254 districts of South Korea from 2015 to 2021. We use 77 districts of Seoul, Incheon, and Gyeonggi among 254 districts. The area of Michuhol-gu was excluded from the analysis from 2015 to 2017 because no deaths were recorded during this period due to measurement errors. Likewise, the area of Bucheon-si was excluded for the year of 2015 for the same reason. We first need to do the data preprocessing.

Now we apply Bayesian hierarchical model, and to do this, we need to convert the data into appropriate format.
```r
# We convert "death77_FINAL" into "ddmultinom".
M <- function(x){
  nc <- dim(x)[2]
  res <- NULL
  for (j in 1:nc)
    res <- cbind(res, x[,j])
  return(res)
}

multinomdata <- function(data = death77_FINAL,
                        X = c("O3_8H_lag0", "temp_stand"),
                        Y = "count",
                        i = "region",
                        ij = "zone_ymd"){
  Xnames <- X
  Yname <- Y
  # Identifier of Unique Zone (i)
  names(data)[which(names(data) == i)] <- "i"
  # Identifier of Unique Zone*Time stratum (i.e. "ij", named here simply "j")
  names(data)[which(names(data) == ij)] <- "j"
  # Select Variables
  nX <- length(Xnames)
  varnames <- c(Yname, Xnames)
  data <- data[, c("i", "j", varnames)]
  # Ordering Data by time stratum order
  data <- data[order(data$j), ]
  rownames(data) <- NULL
  # Reshape for long to wide according to "j" and get "k"
  ks <- as.numeric(table(data$j))
  # Variable time for reshape (k counter within each stratum "j")
  data$k <- unlist(lapply(ks, FUN = function(x) 1:x))
  # Reshape
  dataw <- reshape(data, direction = "wide", idvar = "j", timevar = "k", sep = "")
  dataw$K <- ks
  # Drop observations with K = 1 (i.e. drop zone-time strata with only one observation, which can be the case of having missing data; lost 0f 0.16% of the multinomials (all in zones 13 and 14)
  dataw <- dataw[dataw$K>1, ]
  # Recover "i" and remove copies of "i"
  names(dataw)[names(dataw) == "i1"] <- "i"
  kmax <- max(ks)
  dataw[, paste0("i", 2:kmax)] <- NULL
  # Sort Variables
  dataw <- dataw[, c("i", "j",
                    paste(rep(varnames, kmax),
                          rep(1:kmax, each = length(varnames)),
                          sep=""),
                    "K")]
  # Split the data set according to the size of the multinomial distribution K to, taking into account structural missingness, keep only complete cases (regarding non-structural missing data) and drop multinomial observations with the sum of outcomes count being zero (which do not contribute to the likelihood and hence, to parameters estimation)
  # Split data according to K
  dataws <- split(dataw, dataw$K)
  nK <- length(dataws)
  Ks <- rep(NA, nK)
  for (i in 1:nK) {
    ddi <- dataws[[i]]
    k <- ddi$K[1]
    # Drop strata with 0 counts sum
    nonzero <- rowSums(ddi[, paste0(Yname, 1:k)]) > 0
    ddi <- ddi[nonzero, ]
    # Keep only complete cases
    cc <- complete.cases(ddi[, c("i", "j", paste0(rep(varnames, each = k), 1:k))])
    dataws[[i]] <- ddi[cc, ]
    Ks[i] <- k
  }
  # Merge to a single data.frame
  dataw <- Reduce(function(x,y) merge(x, y, all = TRUE), dataws)
  # Convert "j" to numeric (now and no before to ensure it takes consecutive values)
  dataw$j <- as.numeric(dataw$j)
  # Format data for BUGS model
  resnames <- c("Ymat", "Svector", paste0("X", 1:nX, "mat"), "ivector",
               "Kvector", "N", "I")
  res <- vector("list", length(resnames))
  names(res) <- resnames
 
  Kmax <- max(Ks)
  res[["ivector"]] <- as.numeric(dataw$i)
  res[["Kvector"]] <- as.numeric(dataw$K)
  Ymat <- M(dataw[, paste0(Yname, 1:Kmax)])
  res[["Ymat"]] <- Ymat
  res[["Svector"]] <- rowSums(Ymat, na.rm = TRUE)
  rm(Ymat)
  res[["N"]] <- dim(dataw)[1]
  res[["I"]] <- max(dataw$i)
  for (i in 1:nX)
    res[[paste0("X", i, "mat")]] <- M(dataw[, paste0(Xnames[i], 1:Kmax)])
 
  return(res)
}

ddmultinom = multinomdata(data = data.frame(death77_FINAL), X = Xnames, Y = "count",
                           i = "region", ij = "zone_ymd")
```

```r
# Define Bayesian hierarchical model
bugmultinom = function(){
  for(r in 1:N){
    Ymat[r, 1:Kvector[r]] ~ dmulti(prob[r, 1:Kvector[r]], Svector[r])
    for (s in 1:Kvector[r]) {
      prob[r,s] <- g[r,s]/sum(g[r, 1:Kvector[r]])
      log(g[r,s]) <- beta1 * X1mat[r,s] + u1[ivector[r]] * X1mat[r,s] +
        beta2 * X2mat[r,s]
      }
  }
  for (i in 1:I) {
    u1[i] ~ dnorm(0, tau1)
  }
  beta1 ~ dnorm(5, 1)
  beta2 ~ dnorm(0, 1)
  tau1 ~ dgamma(4, 1)
  sigma1 <- sqrt(1/tau1)
}

# Set initial values
Xnames = c("O3_8H_lag0", "temp_stand")
nX = length(Xnames)
parslabels = c(paste0("beta", 1:nX), "sigma1")
parameters = c(parslabels, "u1")
parsini = list(c(beta=0, tau1=0.1), c(beta=10, tau1=10))
initials= lapply(parsini,
                  FUN=function(cha){
                    res = as.list(c(rep(cha[1], nX), cha[2]))
                    names(res) = c(paste0("beta", 1:nX), "tau1")
                    return(res)
                  })
initials[[1]]$beta2 = 0
initials[[2]]$beta2 = 1

# Set Iteration
Iter = 2500; Thin = 2; Burn = 500; Chain = 2

# Implement Bayesian posterior inference
modmultinomial <- R2jags::jags(data = ddmultinom,
                              inits = initials,
                              parameters.to.save = parameters,
                              model.file = bugmultinom,
                              n.iter = (Iter * Thin + Burn),
                              n.burnin = Burn,
                              n.thin = Thin,
                              n.chains = Chain,
                              DIC = TRUE)
```
